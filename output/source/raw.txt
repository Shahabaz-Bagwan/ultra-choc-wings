units:
  # Size of each key. cx and cy for choc, u and U for MX.
  kx: 16 
  ky: 16
  # key spread and padding
  # Chocofi spacing: 18 mm columns, 17 mm rows (Kailh Choc spacing)
  px: 18
  py: 17
  # From that, get the spacing between keys
  ksx: px-kx
  ksy: py-ky
  lpx: px + 1
  
  # Column stagger, in mm, copied from the Chocofi PCB (pashutk/chocofi),
  # which keeps the Fifi stagger in mm with the pinky moved down.
  stagger1: 12      # ring, above pinky
  stagger2: 6       # middle, above ring
  stagger3: -6.625  # index, below middle
  stagger4: -2.375  # inner, below index

  # CR2032 holder position, relative to the pinky bottom key
  coin_x: 22
  coin_y: -10

  # M2 case mounting holes, relative to the pinky bottom key

  hole_top_y: 59.5
  hole_bottom_x: 2
  hole_bottom_y: -10.8
  hole_thumb_x: 76.5
  hole_thumb_y: -2.5

  # Board outline (see outlines.board_noround): end of the flat top edge,

  bottom_x: 21
  bottom_y: -26.5
  bottom_left_y: -10
  corner_r: 3

  # Board top edge above the pinky top key (clears the XIAO's top pin row)
  mcu_top: 28.9

  xiao_length: 21
  xiao_width: 17.78

    footprints:
      # Every footprint below is reversible: the same PCB is used for the
      # left half (parts on F) and the right half (parts on B). All SMD parts
      # sit on the switch side so JLCPCB only has to assemble one side per half.
      pg1316s_switch:
        what: pg1316s
        where: true
        params:
          from: "{{colrow}}"
          to: "{{column_net}}"
      # Matrix diode, in the strip below the switch body (under the keycap).
      diode:
        what: rev_smd2
        where: /matrix_/
        params:
          designator: D
          package: SOD123
          polarized: true
          value: 1N4148W
          pad1: "{{row_net}}"
          pad2: "{{colrow}}"
        adjust:
          shift: [2.6, -7.2]
      # Thumb keys sit on the board edge, so their diode and resistor go in the
      # strip above the switch instead.
      diode_thumb:
        what: rev_smd2
        where: /thumbfan_/
        params:
          designator: D
          package: SOD123
          polarized: true
          value: 1N4148W

          pad2: "{{colrow}}"
        adjust:
          shift: [1.3, 7.2]
      # Per-key LED in the component pocket under the PG1316S body, with its
